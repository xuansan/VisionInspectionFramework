if(NOT TARGET vision_application_qt OR NOT TARGET Qt6::Widgets)
    message(FATAL_ERROR "Configure the framework with VISION_EXTENSION_DIR pointing here; see APPLICATION.md")
endif()
add_executable(@PROJECT_NAME@ main.cpp)
vision_demo_resources(@PROJECT_NAME@)
target_link_libraries(@PROJECT_NAME@ PRIVATE vision_serialization Qt6::Widgets)
vision_configure_target(@PROJECT_NAME@)
add_custom_command(TARGET @PROJECT_NAME@ POST_BUILD
    COMMAND ${CMAKE_COMMAND} -E copy_if_different "${CMAKE_CURRENT_SOURCE_DIR}/recipe.json" "$<TARGET_FILE_DIR:@PROJECT_NAME@>/recipe.json")
foreach(component IN ITEMS Gui Widgets)
    add_custom_command(TARGET @PROJECT_NAME@ POST_BUILD COMMAND ${CMAKE_COMMAND} -E copy_if_different "$<TARGET_FILE:Qt6::${component}>" "$<TARGET_FILE_DIR:@PROJECT_NAME@>")
endforeach()
foreach(plugin IN ITEMS QWindowsIntegrationPlugin QOffscreenIntegrationPlugin)
    add_custom_command(TARGET @PROJECT_NAME@ POST_BUILD
        COMMAND ${CMAKE_COMMAND} -E make_directory "$<TARGET_FILE_DIR:@PROJECT_NAME@>/platforms"
        COMMAND ${CMAKE_COMMAND} -E copy_if_different "$<TARGET_FILE:Qt6::${plugin}>" "$<TARGET_FILE_DIR:@PROJECT_NAME@>/platforms")
endforeach()
file(GENERATE OUTPUT "${CMAKE_CURRENT_BINARY_DIR}/$<CONFIG>/qt.conf" CONTENT "[Paths]\nPlugins=.\n")
if(BUILD_TESTING)
    foreach(scenario IN ITEMS normal ng stop restart)
        add_test(NAME user.@PROJECT_NAME@.${scenario} COMMAND @PROJECT_NAME@ --smoke ${scenario})
        set_tests_properties(user.@PROJECT_NAME@.${scenario} PROPERTIES TIMEOUT 20 ENVIRONMENT "QT_QPA_PLATFORM=offscreen")
    endforeach()
endif()
