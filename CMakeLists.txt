if(VISION_BUILD_GUI OR VISION_BUILD_LOCAL_IPC)

        set(Qt6_DIR "${CMAKE_CURRENT_SOURCE_DIR}/开发环境/SDK/Qt/6.8.3/msvc2022_64/lib/cmake/Qt6")
    endif()
    # This development skeleton deploys its DLLs explicitly, not via Qt's
    # install-time deployment API. Qt 6.8.3 deployment setup with this local
    # CMake 3.29.5-msvc4 crashes on repeat configuration; see VAL-20260918-004.
    # Reassess this pinned-Qt internal switch when implementing T31 packaging.
    set(QT_SKIP_SETUP_DEPLOYMENT ON)
    set(_vision_qt_components Core)
    if(VISION_BUILD_GUI)
        list(APPEND _vision_qt_components Gui Widgets)
    endif()
    if(VISION_BUILD_LOCAL_IPC)
        list(APPEND _vision_qt_components Network)
    endif()
    find_package(Qt6 6.8.3 EXACT REQUIRED COMPONENTS ${_vision_qt_components})
endif()
